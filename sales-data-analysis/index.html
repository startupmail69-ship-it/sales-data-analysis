SQL Queries Used

Business questions solved using SQL for sales, customer, product, and shipping analysis.

💰 Revenue by Department & Quarter


SELECT
    department,
    quarter,
    SUM(revenue) AS total_revenue
FROM sales_data
GROUP BY department, quarter
ORDER BY total_revenue DESC;

🏆 Top 5 Products by Revenue


SELECT
    product,
    SUM(revenue) AS total_revenue
FROM sales_data
GROUP BY product
ORDER BY total_revenue DESC
LIMIT 5;

👥 Top 3 Customers


SELECT
    customer,
    COUNT(*) AS completed_orders
FROM sales_data
WHERE order_status = 'Completed'
GROUP BY customer
ORDER BY completed_orders DESC
LIMIT 3;

🚚 Average Shipping Delay


SELECT
    shipping_mode,
    AVG(shipping_delay) AS avg_delay
FROM sales_data
GROUP BY shipping_mode;

❌ Order Cancellation Percentage


SELECT
    100.0 *
    SUM(
        CASE
            WHEN order_status = 'Cancelled'
            THEN 1 ELSE 0
        END
    ) / COUNT(*) AS cancellation_percentage
FROM sales_data;

📦 Products Never Ordered


SELECT
    p.product_name
FROM product_info p
LEFT JOIN ordered_items o
    ON p.product_id = o.product_id
WHERE o.product_id IS NULL;
